**Template design rationale**:

1. **Section 1 (핵심 요약)**: 30 seconds to read. One table, one sentence. The user decides immediately whether to read further.
2. **Section 2 (섹터별 방향)**: The actionable core. Each sector gets a direction call with top 5 supporting signals. If the user owns semiconductor stocks, they scroll directly to IT sector.
3. **Section 3 (Top 10)**: For the curious reader who wants to see the strongest individual signals regardless of sector.
4. **Section 4 (증거 체인)**: The trust mechanism. Shows convergent multi-source evidence visually. Without this, InvestScan is no different from a news aggregator.
5. **Section 5 (주의 사항)**: Legal protection and intellectual honesty. Low-confidence sectors explicitly flagged.

**What is NOT in the template**: Historical comparison ("vs last week"), HTML rendering, confidence heatmaps, radar charts. These are Branch 3.2 territory.

#### 3.1.2 Report Generation Code: `generate_report.py`

```python
"""InvestScan Report Generator -- Minimal Implementation.

Reads synthesis output (JSON), renders Jinja2 template, writes Markdown.
~250 LOC including helpers.
"""

from __future__ import annotations

import json
from dataclasses import dataclass, field
from datetime import date, datetime
from pathlib import Path
from typing import Any

from jinja2 import Environment, FileSystemLoader, select_autoescape


# ── Data Transfer Objects (frozen, from synthesis stage) ──────────────

@dataclass(frozen=True)
class SignalView:
    """Read-only view of a signal for template rendering."""
    title: str
    steeps_category: str
    signal_layer: str
    source_system: str
    confidence: float
    direction: str          # "bullish" | "bearish" | "neutral"
    direction_ko: str       # "강세" | "약세" | "중립"
    conviction: float       # 0-1
    sectors: tuple[str, ...]
    source_count: int
    evidence_summary: str


@dataclass(frozen=True)
class SectorView:
    """Sector-level aggregation for template."""
    name_en: str
    name_ko: str
    direction: str
    direction_ko: str
    conviction: float
    signal_count: int
    signals: tuple[SignalView, ...]
    rationale: str


@dataclass(frozen=True)
class EvidenceChainView:
    """Cross-source convergence evidence."""
    theme: str
    source_1: str
    source_2: str
    source_3: str | None
    source_count: int
    conclusion: str
    affected_sectors: tuple[str, ...]


@dataclass(frozen=True)
class DirectionSummary:
    """Aggregate direction counts."""
    direction: str
    direction_ko: str
    count: int
    avg_conviction: float


@dataclass(frozen=True)
class WarningView:
    level: str              # "HIGH" | "MEDIUM" | "LOW"
    message: str


@dataclass(frozen=True)
class ReportData:
    """Complete data package for template rendering.

    This is the ONLY object the template receives. All computation
    happens BEFORE this object is constructed. The template does
    zero logic -- only iteration and formatting.
    """
    report_date_range: str          # "2026-03-21 ~ 2026-03-27"
    generated_at: str               # ISO timestamp
    envscan_count: int
    gnews_count: int
    total_signals: int
    dominant_direction: str         # "강세(Bullish)"
    direction_summary: tuple[DirectionSummary, ...]
    top_signal: SignalView
    sectors: tuple[SectorView, ...]
    top_signals: tuple[SignalView, ...]
    evidence_chains: tuple[EvidenceChainView, ...]
    warnings: tuple[WarningView, ...]
    low_confidence_sectors: tuple[str, ...]
    version: str


# ── Direction mapping ────────────────────────────────────────────────

DIRECTION_KO = {
    "bullish": "강세(Bullish)",
    "bearish": "약세(Bearish)",
    "neutral": "중립(Neutral)",
}


# ── Builder: synthesis JSON -> ReportData ────────────────────────────

def _build_signal_view(raw: dict[str, Any]) -> SignalView:
    direction = raw.get("direction", "neutral")
    return SignalView(
        title=raw["title"],
        steeps_category=raw.get("steeps_category", "N/A"),
        signal_layer=raw.get("signal_layer", "N/A"),
        source_system=raw.get("source_system", "unknown"),
        confidence=raw.get("confidence", 0.0),
        direction=direction,
        direction_ko=DIRECTION_KO.get(direction, "중립"),
        conviction=raw.get("conviction", 0.0),
        sectors=tuple(raw.get("sectors", [])),
        source_count=raw.get("source_count", 1),
        evidence_summary=raw.get("evidence_summary", ""),
    )


def _build_sector_view(name_en: str, name_ko: str, signals: list[dict]) -> SectorView:
    sig_views = tuple(_build_signal_view(s) for s in signals)
    if not sig_views:
        return SectorView(
            name_en=name_en, name_ko=name_ko,
            direction="neutral", direction_ko="중립",
            conviction=0.0, signal_count=0,
            signals=(), rationale="신호 없음",
        )
    # Direction = majority vote weighted by conviction
    dir_scores: dict[str, float] = {}
    for sv in sig_views:
        dir_scores[sv.direction] = dir_scores.get(sv.direction, 0.0) + sv.conviction
    dominant = max(dir_scores, key=dir_scores.get)  # type: ignore[arg-type]
    avg_conv = sum(sv.conviction for sv in sig_views) / len(sig_views)
    return SectorView(
        name_en=name_en, name_ko=name_ko,
        direction=dominant,
        direction_ko=DIRECTION_KO.get(dominant, "중립"),
        conviction=avg_conv,
        signal_count=len(sig_views),
        signals=tuple(sorted(sig_views, key=lambda s: s.conviction, reverse=True)),
        rationale=_generate_sector_rationale(sig_views),
    )


def _generate_sector_rationale(signals: tuple[SignalView, ...]) -> str:
    """One-sentence rationale from top signals. Deterministic, no LLM."""
    if not signals:
        return ""
    top = signals[0]
    count = len(signals)
    sources = {s.source_system for s in signals}
    source_text = "양대 소스 수렴" if len(sources) > 1 else f"{top.source_system} 단일 소스"
    return (
        f"{count}개 신호 기반 ({source_text}). "
        f"최강 신호: {top.title[:50]} (확신도 {top.conviction:.0%})"
    )


def _build_evidence_chains(synthesis: dict[str, Any]) -> tuple[EvidenceChainView, ...]:
    """Extract convergence clusters from synthesis output."""
    chains = []
    for cluster in synthesis.get("convergence_clusters", []):
        sources = cluster.get("sources", [])
        chains.append(EvidenceChainView(
            theme=cluster.get("theme", "Unknown"),
            source_1=sources[0] if len(sources) > 0 else "",
            source_2=sources[1] if len(sources) > 1 else "",
            source_3=sources[2] if len(sources) > 2 else None,
            source_count=len(sources),
            conclusion=cluster.get("conclusion", ""),
            affected_sectors=tuple(cluster.get("sectors", [])),
        ))
    return tuple(chains)


def build_report_data(synthesis_path: Path, version: str = "0.1.0") -> ReportData:
    """Read synthesis JSON and construct ReportData.

    This function is the SOLE bridge between the synthesis stage
    and the report template. All data transformation happens here.
    """
    with open(synthesis_path, encoding="utf-8") as f:
        synthesis = json.load(f)

    # Build signal views
    raw_signals = synthesis.get("signals", [])
    all_signals = [_build_signal_view(s) for s in raw_signals]
    all_signals.sort(key=lambda s: s.conviction, reverse=True)

    # Partition by source
    envscan_count = sum(1 for s in all_signals if s.source_system == "envscan")
    gnews_count = sum(1 for s in all_signals if s.source_system == "gnews")

    # Build sector views
    sector_map: dict[str, list[dict]] = {}
    sector_names: dict[str, str] = synthesis.get("sector_name_map", {})
    for raw_sig in raw_signals:
        for sector in raw_sig.get("sectors", []):
            sector_map.setdefault(sector, []).append(raw_sig)
    sector_views = tuple(
        _build_sector_view(en, sector_names.get(en, en), sigs)
        for en, sigs in sorted(sector_map.items())
    )

    # Direction summary
    dir_counts: dict[str, list[float]] = {}
    for s in all_signals:
        dir_counts.setdefault(s.direction, []).append(s.conviction)
    direction_summary = tuple(
        DirectionSummary(
            direction=d,
            direction_ko=DIRECTION_KO.get(d, "중립"),
            count=len(convs),
            avg_conviction=sum(convs) / len(convs) if convs else 0.0,
        )
        for d, convs in sorted(dir_counts.items(), key=lambda x: -len(x[1]))
    )

    # Dominant direction
    dominant = direction_summary[0].direction_ko if direction_summary else "중립"

    # Warnings
    warnings = []
    low_conf_sectors = []
    for sv in sector_views:
        if sv.conviction < 0.5:
            low_conf_sectors.append(sv.name_ko)
    single_source = [s for s in all_signals if s.source_count == 1 and s.conviction > 0.7]
    if single_source:
        warnings.append(WarningView(
            level="MEDIUM",
            message=f"단일 소스 고확신 신호 {len(single_source)}건 -- 추가 검증 권장",
        ))
    if envscan_count == 0:
        warnings.append(WarningView(
            level="HIGH",
            message="EnvironmentScan 데이터 없음 -- GlobalNews 단일 소스 기반 분석",
        ))
    if gnews_count == 0:
        warnings.append(WarningView(
            level="HIGH",
            message="GlobalNews 데이터 없음 -- EnvironmentScan 단일 소스 기반 분석",
        ))

    # Date range
    date_range = synthesis.get("date_range", "N/A")

    return ReportData(
        report_date_range=date_range,
        generated_at=datetime.now().isoformat(timespec="seconds"),
        envscan_count=envscan_count,
        gnews_count=gnews_count,
        total_signals=len(all_signals),
        dominant_direction=dominant,
        direction_summary=direction_summary,
        top_signal=all_signals[0] if all_signals else _empty_signal(),
        sectors=sector_views,
        top_signals=tuple(all_signals[:10]),
        evidence_chains=_build_evidence_chains(synthesis),
        warnings=tuple(warnings),
        low_confidence_sectors=tuple(low_conf_sectors),
        version=version,
    )


def _empty_signal() -> SignalView:
    return SignalView(
        title="(신호 없음)", steeps_category="N/A", signal_layer="N/A",
        source_system="none", confidence=0.0, direction="neutral",
        direction_ko="중립", conviction=0.0, sectors=(), source_count=0,
        evidence_summary="",
    )


# ── Renderer ─────────────────────────────────────────────────────────

def generate_report(
    synthesis_path: Path,
    template_dir: Path | None = None,
    output_dir: Path | None = None,
    version: str = "0.1.0",
) -> Path:
    """Generate weekly investment report from synthesis output.

    Args:
        synthesis_path: Path to investment_synthesis.json
        template_dir: Directory containing weekly-report.md.j2
        output_dir: Where to write the report. Defaults to synthesis_path.parent.
        version: InvestScan version string.

    Returns:
        Path to generated report file.
    """
    if template_dir is None:
        template_dir = Path(__file__).parent / "templates"
    if output_dir is None:
        output_dir = synthesis_path.parent

    report_data = build_report_data(synthesis_path, version)

    env = Environment(
        loader=FileSystemLoader(str(template_dir)),
        autoescape=select_autoescape([]),  # Markdown, no HTML escaping
        trim_blocks=True,
        lstrip_blocks=True,
    )
    template = env.get_template("weekly-report.md.j2")

    # Render: template receives a single flat namespace from the frozen dataclass
    from dataclasses import asdict
    rendered = template.render(**asdict(report_data))

    # Write output
    output_dir.mkdir(parents=True, exist_ok=True)
    report_file = output_dir / f"invest-report-{date.today().isoformat()}.md"
    report_file.write_text(rendered, encoding="utf-8")

    return report_file


# ── CLI entry point ──────────────────────────────────────────────────

if __name__ == "__main__":
    import sys
    if len(sys.argv) < 2:
        print("Usage: python generate_report.py <synthesis.json> [output_dir]")
        sys.exit(1)
    synthesis = Path(sys.argv[1])
    out = Path(sys.argv[2]) if len(sys.argv) > 2 else None
    result = generate_report(synthesis, output_dir=out)
    print(f"Report generated: {result}")
```

**LOC count**: ~250 Python + ~70 Jinja2 = ~320 LOC total.

**Key design decisions**:

1. **ReportData is frozen**: The template receives an immutable, fully-computed data object. Zero computation in the template. This means template bugs cannot corrupt data, and data bugs cannot be hidden in template logic.

2. **Direction majority vote weighted by conviction**: A sector with 3 weak bullish signals and 1 strong bearish signal is correctly classified as bearish if the bearish conviction outweighs the three bullish ones combined.

3. **Defensive everywhere**: `raw.get()` with defaults on every field. Missing data produces degraded output, not crashes. A report with "(신호 없음)" is better than no report.

4. **No LLM in minimal**: The `_generate_sector_rationale()` is a deterministic string builder. Reproducible runs. Zero API cost. The Ollama hybrid narrative is Branch 3.2 territory.

**Readability in 5 minutes**: The report is structured for progressive depth. A user can read Section 1 (30 sec) and stop. Or continue to Section 2 (2 min) for sector-level detail. Sections 3-5 are for deep dives. No section requires reading a previous section to be understood.

#### 3.1.3 Decision Journal: Minimal Implementation

**Storage choice: JSON Lines (`.jsonl`), not SQLite.**

Rationale for JSON Lines over SQLite in minimal:
- Zero dependency (no `sqlite3` import needed, but it is stdlib so negligible)
- Human-readable: `cat journal.jsonl | python -m json.tool` works
- Append-only by nature: `open(path, "a")` -- no transactions needed
- For <500 entries (2+ years of weekly decisions), linear scan is <10ms
- Migrating to SQLite later costs ~30 minutes of work

```python
"""InvestScan Decision Journal -- Minimal Append-Only Implementation.

~150 LOC. JSONL storage. No external dependencies.
"""

from __future__ import annotations

import json
from dataclasses import dataclass, field, asdict
from datetime import date, datetime, timedelta
from pathlib import Path
from typing import Any
from uuid import uuid4


# ── Schema ───────────────────────────────────────────────────────────

@dataclass(frozen=True)
class JournalEntry:
    """A single investment decision record.

    Fields:
        entry_id: Unique ID (UUID4)
        created_at: When the decision was recorded
        signal_ids: InvestScan signal IDs that informed this decision
        decision: What action was taken ("Overweight IT sector by 5%")
        rationale: Why this decision was made
        confidence: Self-assessed confidence ("high" | "medium" | "low")
        review_date: When to check the outcome (default: 90 days)
        outcome: Filled in later during review (None until reviewed)
        outcome_date: When outcome was recorded
        outcome_notes: Free-text reflection on what actually happened
    """
    entry_id: str
    created_at: str                     # ISO date
    signal_ids: tuple[str, ...]
    decision: str
    rationale: str
    confidence: str                     # "high" | "medium" | "low"
    review_date: str                    # ISO date
    outcome: str | None = None          # "correct" | "incorrect" | "mixed" | "inconclusive"
    outcome_date: str | None = None
    outcome_notes: str | None = None


# ── Journal Operations ───────────────────────────────────────────────

JOURNAL_FILE = "decision_journal.jsonl"


def _journal_path(data_dir: Path) -> Path:
    return data_dir / JOURNAL_FILE


def journal_add(
    data_dir: Path,
    decision: str,
    signal_ids: list[str],
    rationale: str,
    confidence: str = "medium",
    review_days: int = 90,
) -> JournalEntry:
    """Append a new decision to the journal.

    Args:
        data_dir: Directory where journal.jsonl lives.
        decision: Description of the investment action taken.
        signal_ids: InvestScan signal IDs that informed this decision.
        rationale: Why this decision was made.
        confidence: "high", "medium", or "low".
        review_days: Days until outcome review (default 90).

    Returns:
        The created JournalEntry.
    """
    if confidence not in ("high", "medium", "low"):
        raise ValueError(f"confidence must be high/medium/low, got: {confidence}")

    today = date.today()
    entry = JournalEntry(
        entry_id=str(uuid4())[:8],
        created_at=today.isoformat(),
        signal_ids=tuple(signal_ids),
        decision=decision,
        rationale=rationale,
        confidence=confidence,
        review_date=(today + timedelta(days=review_days)).isoformat(),
    )

    path = _journal_path(data_dir)
    path.parent.mkdir(parents=True, exist_ok=True)
    with open(path, "a", encoding="utf-8") as f:
        f.write(json.dumps(asdict(entry), ensure_ascii=False) + "\n")

    return entry


def _read_all(data_dir: Path) -> list[dict[str, Any]]:
    """Read all journal entries."""
    path = _journal_path(data_dir)
    if not path.exists():
        return []
    entries = []
    with open(path, encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if line:
                entries.append(json.loads(line))
    return entries


def journal_review(
    data_dir: Path,
    days: int = 90,
    include_reviewed: bool = False,
) -> list[dict[str, Any]]:
    """Return decisions that are due for outcome review.

    A decision is "due" if today >= review_date and outcome is None.

    Args:
        data_dir: Directory where journal.jsonl lives.
        days: Not used for filtering; kept for CLI compatibility.
              All past-due entries are returned regardless.
        include_reviewed: If True, also include entries that already have outcomes.

    Returns:
        List of journal entries due for review.
    """
    today = date.today().isoformat()
    entries = _read_all(data_dir)
    due = []
    for entry in entries:
        if entry.get("review_date", "9999-12-31") <= today:
            if include_reviewed or entry.get("outcome") is None:
                due.append(entry)
    return due


def journal_record_outcome(
    data_dir: Path,
    entry_id: str,
    outcome: str,
    notes: str = "",
) -> bool:
    """Record the outcome of a past decision.

    Since JSONL is append-only, this rewrites the entire file.
    Acceptable for <500 entries. For larger journals, migrate to SQLite.

    Args:
        entry_id: The entry_id to update.
        outcome: "correct" | "incorrect" | "mixed" | "inconclusive"
        notes: Free-text reflection.

    Returns:
        True if entry was found and updated, False otherwise.
    """
    valid_outcomes = ("correct", "incorrect", "mixed", "inconclusive")
    if outcome not in valid_outcomes:
        raise ValueError(f"outcome must be one of {valid_outcomes}, got: {outcome}")

    path = _journal_path(data_dir)
    entries = _read_all(data_dir)
    found = False
    for entry in entries:
        if entry.get("entry_id") == entry_id:
            entry["outcome"] = outcome
            entry["outcome_date"] = date.today().isoformat()
            entry["outcome_notes"] = notes
            found = True
            break

    if found:
        with open(path, "w", encoding="utf-8") as f:
            for entry in entries:
                f.write(json.dumps(entry, ensure_ascii=False) + "\n")

    return found


def journal_list(data_dir: Path, limit: int = 20) -> list[dict[str, Any]]:
    """List recent journal entries (newest first)."""
    entries = _read_all(data_dir)
    entries.sort(key=lambda e: e.get("created_at", ""), reverse=True)
    return entries[:limit]


def journal_stats(data_dir: Path) -> dict[str, Any]:
    """Basic statistics on journal entries."""
    entries = _read_all(data_dir)
    if not entries:
        return {"total": 0}

    reviewed = [e for e in entries if e.get("outcome")]
    correct = [e for e in reviewed if e["outcome"] == "correct"]

    return {
        "total": len(entries),
        "pending_review": len([e for e in entries if e.get("outcome") is None]),
        "reviewed": len(reviewed),
        "accuracy": len(correct) / len(reviewed) if reviewed else None,
        "by_confidence": {
            conf: len([e for e in entries if e.get("confidence") == conf])
            for conf in ("high", "medium", "low")
        },
    }
```

```python
"""Rich report additions -- confidence heatmap, historical comparison, HTML output.

These functions extend the minimal ReportData with additional computed views.
~200 additional LOC on top of the minimal 250 LOC.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any


# ── Confidence Heatmap ───────────────────────────────────────────────

STEEPS_DIMS = ("S", "T", "E", "E_env", "P", "s")

HEATMAP_SYMBOLS = {
    "high_bull": "▲▲",
    "bull": "▲",
    "neutral": "—",
    "bear": "▼",
    "high_bear": "▼▼",
    "empty": "·",
}


@dataclass(frozen=True)
class HeatmapRow:
    sector_ko: str
    S: float | None
    T: float | None
    E: float | None
    E_env: float | None
    P: float | None
    s: float | None
    total: str           # Symbol representation
    total_val: float     # Numeric for HTML coloring


def _score_to_symbol(val: float | None) -> str:
    if val is None:
        return HEATMAP_SYMBOLS["empty"]
    if val > 0.70:
        return HEATMAP_SYMBOLS["high_bull"]
    if val > 0.50:
        return HEATMAP_SYMBOLS["bull"]
    if val > 0.30:
        return HEATMAP_SYMBOLS["neutral"]
    if val > 0.15:
        return HEATMAP_SYMBOLS["bear"]
    return HEATMAP_SYMBOLS["high_bear"]


def build_confidence_heatmap(
    sectors: tuple,  # tuple[SectorView, ...]
    all_signals: list,
) -> tuple[HeatmapRow, ...]:
    """Build sector x STEEPs confidence matrix.

    For each (sector, STEEPs dimension) pair, compute the average
    conviction of signals matching both. None if no signals match.
    """
    # Index: (sector_en, steeps) -> list[conviction]
    matrix: dict[tuple[str, str], list[float]] = {}
    for sig in all_signals:
        for sector in (sig.sectors if hasattr(sig, 'sectors') else sig.get("sectors", [])):
            steeps = sig.steeps_category if hasattr(sig, 'steeps_category') else sig.get("steeps_category", "")
            # Map full names to short codes
            dim = _steeps_to_dim(steeps)
            if dim:
                conv = sig.conviction if hasattr(sig, 'conviction') else sig.get("conviction", 0.0)
                matrix.setdefault((sector, dim), []).append(conv)

    rows = []
    for sv in sectors:
        dim_vals = {}
        for dim in STEEPS_DIMS:
            convs = matrix.get((sv.name_en, dim), [])
            dim_vals[dim] = sum(convs) / len(convs) if convs else None

        non_none = [v for v in dim_vals.values() if v is not None]
        total_val = sum(non_none) / len(non_none) if non_none else 0.0

        rows.append(HeatmapRow(
            sector_ko=sv.name_ko,
            S=dim_vals["S"],
            T=dim_vals["T"],
            E=dim_vals["E"],
            E_env=dim_vals["E_env"],
            P=dim_vals["P"],
            s=dim_vals["s"],
            total=_score_to_symbol(total_val),
            total_val=total_val,
        ))

    return tuple(rows)


def _steeps_to_dim(steeps: str) -> str | None:
    """Map STEEPs category strings to dimension codes."""
    mapping = {
        "S": "S", "S_Social": "S",
        "T": "T", "T_Technological": "T",
        "E": "E", "E_Economic": "E",
        "E_env": "E_env", "E_Environmental": "E_env",
        "P": "P", "P_Political": "P",
        "s": "s", "s_security": "s",
    }
    return mapping.get(steeps)


# ── Historical Comparison ────────────────────────────────────────────

def load_previous_week(output_dir, current_date) -> dict[str, Any] | None:
    """Load last week's synthesis for comparison.

    Looks for output/{prev_date}/investment_synthesis.json.
    Returns dict[sector_en -> {direction, direction_ko, conviction}] or None.
    """
    from datetime import timedelta
    import json

    prev_date = current_date - timedelta(days=7)
    prev_path = output_dir / prev_date.isoformat() / "investment_synthesis.json"
    if not prev_path.exists():
        return None

    try:
        with open(prev_path, encoding="utf-8") as f:
            prev = json.load(f)
        result = {}
        for sig in prev.get("signals", []):
            for sector in sig.get("sectors", []):
                if sector not in result:
                    result[sector] = {
                        "direction": sig.get("direction", "neutral"),
                        "direction_ko": DIRECTION_KO.get(
                            sig.get("direction", "neutral"), "중립"
                        ),
                        "conviction": sig.get("conviction", 0.0),
                    }
        return result
    except (json.JSONDecodeError, OSError):
        return None


# ── Narrative Enhancement (Optional Ollama) ──────────────────────────

def generate_narrative_llm(
    summary_data: dict[str, Any],
    model: str = "qwen2.5:32b",
    timeout: int = 120,
) -> str | None:
    """Generate narrative summary via local Ollama.

    Returns None if Ollama is not available or times out.
    Falls back gracefully -- the template renders fine without narrative.
    """
    try:
        import subprocess
        import json

        prompt = (
            "아래 투자 방향 데이터를 기반으로 300자 내외의 한국어 주간 투자 브리핑을 작성하세요. "
            "구체적 수치를 인용하고, 어떤 섹터에 주의해야 하는지 명확히 서술하세요. "
            "투자 조언이 아닌 환경 분석 요약임을 유지하세요.\n\n"
            f"데이터:\n{json.dumps(summary_data, ensure_ascii=False, indent=2)}"
        )

        result = subprocess.run(
            ["ollama", "run", model],
            input=prompt,
            capture_output=True,
            text=True,
            timeout=timeout,
        )

        if result.returncode == 0 and result.stdout.strip():
            return result.stdout.strip()
        return None

    except (FileNotFoundError, subprocess.TimeoutExpired, OSError):
```

```python
"""InvestScan Decision Journal -- Rich Implementation.

Pydantic validation, SQLite storage with FTS5, outcome tracking,
signal accuracy retrospective, CSV export.
~280 LOC.
"""

from __future__ import annotations

import csv
import io
import json
import sqlite3
from datetime import date, datetime, timedelta
from pathlib import Path
from typing import Any, Literal
from uuid import uuid4

from pydantic import BaseModel, Field, field_validator


# ── Schema with Pydantic Validation ──────────────────────────────────

class JournalEntry(BaseModel):
    """Validated investment decision record."""

    entry_id: str = Field(default_factory=lambda: str(uuid4())[:8])
    created_at: date = Field(default_factory=date.today)
    signal_ids: list[str] = Field(min_length=0, max_length=50)
    decision: str = Field(min_length=5, max_length=1000)
    rationale: str = Field(min_length=5, max_length=2000)
    confidence: Literal["high", "medium", "low"]
    sectors: list[str] = Field(default_factory=list)
    direction: Literal["bullish", "bearish", "neutral"] | None = None
    review_date: date = Field(default=None)
    outcome: Literal["correct", "incorrect", "mixed", "inconclusive"] | None = None
    outcome_date: date | None = None
    outcome_notes: str | None = None
    market_context: str | None = None    # What the market actually did

    @field_validator("review_date", mode="before")
    @classmethod
    def set_default_review(cls, v, info):
        if v is None:
            created = info.data.get("created_at", date.today())
            if isinstance(created, str):
                created = date.fromisoformat(created)
            return created + timedelta(days=90)
        return v

    class Config:
        frozen = True


# ── SQLite Backend ───────────────────────────────────────────────────

SCHEMA_SQL = """
CREATE TABLE IF NOT EXISTS decisions (
    entry_id TEXT PRIMARY KEY,
    created_at TEXT NOT NULL,
    signal_ids TEXT NOT NULL,          -- JSON array
    decision TEXT NOT NULL,
    rationale TEXT NOT NULL,
    confidence TEXT NOT NULL,
    sectors TEXT NOT NULL,             -- JSON array
    direction TEXT,
    review_date TEXT NOT NULL,
    outcome TEXT,
    outcome_date TEXT,
    outcome_notes TEXT,
    market_context TEXT
);

CREATE VIRTUAL TABLE IF NOT EXISTS decisions_fts USING fts5(
    decision, rationale, outcome_notes,
    content='decisions',
    content_rowid='rowid'
);

CREATE TRIGGER IF NOT EXISTS decisions_ai AFTER INSERT ON decisions BEGIN
    INSERT INTO decisions_fts(rowid, decision, rationale, outcome_notes)
    VALUES (new.rowid, new.decision, new.rationale, new.outcome_notes);
END;

CREATE TRIGGER IF NOT EXISTS decisions_au AFTER UPDATE ON decisions BEGIN
    INSERT INTO decisions_fts(decisions_fts, rowid, decision, rationale, outcome_notes)
    VALUES ('delete', old.rowid, old.decision, old.rationale, old.outcome_notes);
    INSERT INTO decisions_fts(rowid, decision, rationale, outcome_notes)
    VALUES (new.rowid, new.decision, new.rationale, new.outcome_notes);
END;
"""


class JournalDB:
    """SQLite-backed decision journal with FTS5 search."""

    def __init__(self, db_path: Path):
        self.db_path = db_path
        db_path.parent.mkdir(parents=True, exist_ok=True)
        self.conn = sqlite3.connect(str(db_path))
        self.conn.row_factory = sqlite3.Row
        self.conn.executescript(SCHEMA_SQL)

    def add(self, entry: JournalEntry) -> str:
        """Insert a validated journal entry. Returns entry_id."""
        self.conn.execute(
            """INSERT INTO decisions
               (entry_id, created_at, signal_ids, decision, rationale,
                confidence, sectors, direction, review_date,
                outcome, outcome_date, outcome_notes, market_context)
               VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)""",
            (
                entry.entry_id,
                entry.created_at.isoformat(),
                json.dumps(entry.signal_ids),
                entry.decision,
                entry.rationale,
                entry.confidence,
                json.dumps(entry.sectors),
                entry.direction,
                entry.review_date.isoformat(),
                entry.outcome,
                entry.outcome_date.isoformat() if entry.outcome_date else None,
                entry.outcome_notes,
                entry.market_context,
            ),
        )
        self.conn.commit()
        return entry.entry_id

    def get(self, entry_id: str) -> dict[str, Any] | None:
        row = self.conn.execute(
            "SELECT * FROM decisions WHERE entry_id = ?", (entry_id,)
        ).fetchone()
        return dict(row) if row else None

    def review_due(self, as_of: date | None = None) -> list[dict[str, Any]]:
        """Return entries due for review (outcome is NULL and review_date <= as_of)."""
        as_of = as_of or date.today()
        rows = self.conn.execute(
            """SELECT * FROM decisions
               WHERE outcome IS NULL AND review_date <= ?
               ORDER BY review_date ASC""",
            (as_of.isoformat(),),
        ).fetchall()
        return [dict(r) for r in rows]

    def record_outcome(
        self,
        entry_id: str,
        outcome: str,
        notes: str = "",
        market_context: str = "",
    ) -> bool:
        """Record outcome for a past decision."""
        if outcome not in ("correct", "incorrect", "mixed", "inconclusive"):
            raise ValueError(f"Invalid outcome: {outcome}")
        cur = self.conn.execute(
            """UPDATE decisions
               SET outcome = ?, outcome_date = ?, outcome_notes = ?,
                   market_context = ?
               WHERE entry_id = ?""",
            (outcome, date.today().isoformat(), notes, market_context, entry_id),
        )
        self.conn.commit()
        return cur.rowcount > 0

    def search(self, query: str, limit: int = 20) -> list[dict[str, Any]]:
        """Full-text search across decision, rationale, and outcome notes."""
        rows = self.conn.execute(
            """SELECT d.* FROM decisions d
               JOIN decisions_fts fts ON d.rowid = fts.rowid
               WHERE decisions_fts MATCH ?
               ORDER BY rank
               LIMIT ?""",
            (query, limit),
        ).fetchall()
        return [dict(r) for r in rows]

    def signal_retrospective(self, signal_id: str) -> list[dict[str, Any]]:
        """Find all decisions that referenced a specific signal ID.

        This answers: "Signal IS-20260327-003 -- did decisions based on
        it turn out to be correct?"
        """
        rows = self.conn.execute(
            """SELECT * FROM decisions
               WHERE signal_ids LIKE ?
               ORDER BY created_at DESC""",
            (f'%"{signal_id}"%',),
        ).fetchall()
        return [dict(r) for r in rows]

    def accuracy_by_confidence(self) -> dict[str, dict[str, Any]]:
        """Signal accuracy broken down by confidence level.

        Answers: "Are my high-confidence decisions actually more
        accurate than low-confidence ones?"
        """
        result = {}
        for conf in ("high", "medium", "low"):
            rows = self.conn.execute(
                """SELECT outcome, COUNT(*) as cnt FROM decisions
                   WHERE confidence = ? AND outcome IS NOT NULL
                   GROUP BY outcome""",
                (conf,),
            ).fetchall()
            total = sum(r["cnt"] for r in rows)
            correct = sum(r["cnt"] for r in rows if r["outcome"] == "correct")
            result[conf] = {
                "total_reviewed": total,
                "correct": correct,
                "accuracy": correct / total if total > 0 else None,
                "breakdown": {r["outcome"]: r["cnt"] for r in rows},
            }
        return result

    def export_csv(self) -> str:
        """Export all entries to CSV string."""
        rows = self.conn.execute(
            "SELECT * FROM decisions ORDER BY created_at DESC"
        ).fetchall()
        if not rows:
            return ""
        output = io.StringIO()
        writer = csv.DictWriter(output, fieldnames=rows[0].keys())
        writer.writeheader()
        for row in rows:
            writer.writerow(dict(row))
        return output.getvalue()

    def stats(self) -> dict[str, Any]:
        """Comprehensive journal statistics."""
        total = self.conn.execute("SELECT COUNT(*) FROM decisions").fetchone()[0]
        pending = self.conn.execute(
            "SELECT COUNT(*) FROM decisions WHERE outcome IS NULL"
        ).fetchone()[0]
        reviewed = total - pending

        accuracy = self.accuracy_by_confidence()

        return {
            "total_entries": total,
            "pending_review": pending,
            "reviewed": reviewed,
            "overall_accuracy": (
                sum(a["correct"] for a in accuracy.values())
                / sum(a["total_reviewed"] for a in accuracy.values())
                if any(a["total_reviewed"] > 0 for a in accuracy.values())
                else None
            ),
            "by_confidence": accuracy,
            "oldest_unreviewed": self.conn.execute(
                """SELECT MIN(review_date) FROM decisions
                   WHERE outcome IS NULL"""
            ).fetchone()[0],
        }

    def close(self):
        self.conn.close()
```

### Final Recommendation

**Build Branch 3.1 first, then surgically add two features from Branch 3.2.**

The two features worth borrowing from 3.2:

1. **Confidence heatmap** (+50 LOC in Python, +15 lines in template). The heatmap is the highest-value dense visualization in the entire report. A user can scan all sectors x all STEEPs dimensions in 15 seconds. No other section delivers information this efficiently. Add it to the Markdown template as a simple ASCII table -- no HTML needed.

2. **Journal `signal_retrospective()`** (+20 LOC). The ability to ask "all decisions based on signal IS-20260327-003 -- how did they turn out?" is the core learning mechanism. This can be implemented as a simple linear scan of the JSONL file (no SQLite needed for <500 entries).

**What NOT to build from 3.2**:

- HTML template: The user works in Claude Code terminal and reads Markdown natively (confirmed by architecture analysis). HTML is for sharing with others -- a Phase 2 concern.
- Pydantic for journal: Adds a major dependency for input validation that `if confidence not in (...)` handles in 1 line.
- FTS5 search: With <50 journal entries in year 1, `grep` on the JSONL file is faster than setting up SQLite FTS triggers.
- LLM narrative: Non-deterministic output in financial reports is a net negative until the user has established trust in the deterministic sections (Month 4+).

**Recommended final LOC**: ~505 (minimal) + ~70 (heatmap + retrospective) = **~575 LOC**. Within budget. Maximum value per line.
