# /팀-건강 — FR-22 자가 검증 대시보드

churchTeam 시스템의 건강 상태를 6개 신호로 표시합니다.

```python
import yaml, os, hashlib, json

BASE = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
state_path = os.path.join(BASE, "..", "state.yaml")

with open(state_path) as f:
    state = yaml.safe_load(f)

signals = state.get("churchteam", {}).get("health_signals", {})

print("=" * 50)
print("  churchTeam 자가 검증 카드 (FR-22)")
print("=" * 50)

checks = [
    ("DNA 상속", "dna_inheritance"),
    ("SOT 무결성", "sot_integrity"),
    ("신학 필터", "theology_filter"),
    ("외부 인증", "external_auth"),
    ("자기 진화 이력", "self_evolution_history"),
    ("didim 도달성", "didim_reachability"),
]

for label, key in checks:
    val = signals.get(key, "❓ 미확인")
    print(f"  {label:<16} {val}")

overall = signals.get("overall", "❓ 미확인")
print("=" * 50)
print(f"  종합 상태: {overall}")
print("=" * 50)
```
